    Configuración típica del vhost (orders-app demo)¶

    server {
    listen       80;
    server_name  dist-orders-demo.werfen.com;
    server_tokens on;

    location / {
        root   /srv/www/htdocs/webapps/demo/ordersapp/current/apps/orders/orders-app/public/;
        index  index.php;
        try_files $uri $uri/ /index.php$is_args$args;
    }

    location ~ \.php$ {
        root            .../apps/orders/orders-app/public/;
        fastcgi_pass   127.0.0.1:9000;
        fastcgi_index  index.php;
        fastcgi_param  SCRIPT_FILENAME $document_root$fastcgi_script_name;
        include        fastcgi_params;
        fastcgi_buffers 16 32k;
        fastcgi_buffer_size 64k;
        fastcgi_busy_buffers_size 64k;
        fastcgi_connect_timeout 300;
        fastcgi_send_timeout 300;
        fastcgi_read_timeout 300;
    }
}

    ℹ️ Nota: El vhost de producción (newmyorders.werfen.com) incluye headers de seguridad adicionales: Cache-Control, Content-Security-Policy. El vhost de demo no los tiene.
